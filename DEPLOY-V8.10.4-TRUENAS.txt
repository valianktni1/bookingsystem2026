================================================

This is a cumulative release. Upload its complete contents to the main branch of:

No YAML additions are required.
Deploying sends no client email and does not alter invoice numbers.

1. BACK UP POSTGRESQL
---------------------

BOOKING_COMPOSE=/mnt/apps/dockge/data/bookingsystem2026/compose.yaml
CONTRACT_BACKUP=/mnt/temp-tntermediate/bookingapp2026data/migration-backups/bookingapp-before-full-contract-v8.10.4-2026-08-19.dump

if [ -e "$CONTRACT_BACKUP" ]; then
  echo "STOP: Backup already exists - nothing was overwritten"
else
  docker exec bookingsystem2026-db pg_dump -U bookingapp -d bookingapp -Fc > "$CONTRACT_BACKUP"
fi

ls -lh "$CONTRACT_BACKUP"
docker exec -i bookingsystem2026-db pg_restore -l < "$CONTRACT_BACKUP" | head

2. REBUILD THE APP
------------------

docker compose -f "$BOOKING_COMPOSE" down
docker compose -f "$BOOKING_COMPOSE" build --no-cache app
docker compose -f "$BOOKING_COMPOSE" up -d
