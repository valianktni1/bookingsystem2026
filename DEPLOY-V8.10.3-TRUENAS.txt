1. BACK UP POSTGRESQL
---------------------

BOOKING_COMPOSE=/mnt/apps/dockge/data/bookingsystem2026/compose.yaml
WORKFLOW_BACKUP=/mnt/temp-tntermediate/bookingapp2026data/migration-backups/bookingapp-before-workflow-assurance-2026-08-18.dump

if [ -e "$WORKFLOW_BACKUP" ]; then
  echo "STOP: Backup already exists — nothing was overwritten"
else
  docker exec bookingsystem2026-db pg_dump -U bookingapp -d bookingapp -Fc > "$WORKFLOW_BACKUP"
fi

ls -lh "$WORKFLOW_BACKUP"
docker exec -i bookingsystem2026-db pg_restore -l < "$WORKFLOW_BACKUP" | head

2. REBUILD THE APP
------------------

docker compose -f "$BOOKING_COMPOSE" down
docker compose -f "$BOOKING_COMPOSE" build --no-cache app
docker compose -f "$BOOKING_COMPOSE" up -d
